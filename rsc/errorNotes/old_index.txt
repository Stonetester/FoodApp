        // If JavaScript hasn't initialized after 1 second, show login page
        setTimeout(function() {
            const loginPage = document.getElementById('loginPage');
            const anyPageActive = document.querySelector('.page.active');
            if (loginPage && !anyPageActive) {
                loginPage.classList.add('active');
                console.log('Fallback: Showing login page');
            }
        }, 1000);
        
        // Log any script loading errors
        window.addEventListener('error', function(e) {
            console.error('Script error:', e.filename, e.message, e.lineno);
